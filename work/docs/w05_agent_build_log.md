        "            if any(p in root for p in [\".git\", \"__pycache__\", \".config\", \"sample_data\"]):\n",

        "                rel_path = os.path.relpath(os.path.join(root, f), directory)\n",

        "            \"status\": \"success\",\n",
        "            \"total_project_files\": len(found_files),\n",
